├── client/                    # Flutter 客户端

│   │   │   ├── network/      # 网络
│   │   │   └── utils/        # 工具
│   │   ├── data/             # 数据层
│   │   │   ├── local/        # 本地数据库
│   │   │   ├── models/       # 数据模型
│   │   │   └── repositories/ # 仓库
│   │   └── presentation/     # 表现层
│   │       ├── providers/    # 状态管理
│   │       ├── screens/      # 页面
│   │       └── widgets/      # 组件
│   └── test/                 # 测试文件
│
├── protos/                    # Protobuf 定义
└── .kiro/specs/              # 规范文档

make test-fast
